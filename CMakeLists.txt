cmake_minimum_required(VERSION 3.24)
project(frameeyeosc_panel VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Linux 64-bit is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
include(GNUInstallDirs)
FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.11.3 URL_HASH SHA256=0d8ef5af7f9794e3263480193c491549b2ba6cc74bb018906202ada498a79406)
FetchContent_Declare(openvr URL https://codeload.github.com/ValveSoftware/openvr/tar.gz/41bc3825fd35b04047610c86fee26fb33b017b29 URL_HASH SHA256=8af57b6a016e4037c27a8dce5d5a21eef4bbd5d9efc7f4853187d805a436932f SOURCE_SUBDIR unused)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json openvr)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  set(OPENVR_ARCH linux64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  set(OPENVR_ARCH linuxarm64)
else()
  message(FATAL_ERROR "Unsupported OpenVR processor: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
set(OPENVR_LIBRARY "${openvr_SOURCE_DIR}/lib/${OPENVR_ARCH}/libopenvr_api.so" CACHE FILEPATH "Official OpenVR SDK/runtime library for the target architecture")
add_library(openvr_api SHARED IMPORTED)
set_target_properties(openvr_api PROPERTIES IMPORTED_NO_SONAME TRUE IMPORTED_LOCATION "${OPENVR_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${openvr_SOURCE_DIR}/headers")
add_library(panel_core src/child.cpp src/ipc_client.cpp src/overlay.cpp)
target_link_libraries(panel_core PUBLIC nlohmann_json::nlohmann_json openvr_api)
target_compile_options(panel_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(frameeyeosc-panel src/main.cpp)
target_link_libraries(frameeyeosc-panel PRIVATE panel_core)
set_target_properties(frameeyeosc-panel PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}/frameeyeosc-panel")
set(PANEL_BACKEND "" CACHE FILEPATH "Absolute installed frameeyeosc backend path for the manifest")
set(PANEL_CONTROL_SOCKET "" CACHE FILEPATH "Absolute private control socket for the manifest")
set(PANEL_CONFIG "" CACHE FILEPATH "Optional absolute backend config path for the manifest")
install(TARGETS frameeyeosc-panel RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES "${OPENVR_LIBRARY}" DESTINATION ${CMAKE_INSTALL_LIBDIR}/frameeyeosc-panel)
install(FILES "${openvr_SOURCE_DIR}/LICENSE" DESTINATION ${CMAKE_INSTALL_DATADIR}/frameeyeosc-panel/licenses RENAME OpenVR-LICENSE)
install(FILES "${json_SOURCE_DIR}/LICENSE.MIT" DESTINATION ${CMAKE_INSTALL_DATADIR}/frameeyeosc-panel/licenses RENAME nlohmann-json-LICENSE)
if((PANEL_BACKEND OR PANEL_CONTROL_SOCKET OR PANEL_CONFIG) AND NOT (PANEL_BACKEND AND PANEL_CONTROL_SOCKET))
  message(FATAL_ERROR "Manifest generation requires both PANEL_BACKEND and PANEL_CONTROL_SOCKET; PANEL_CONFIG is optional")
endif()
if(PANEL_BACKEND AND PANEL_CONTROL_SOCKET)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/assets/generate_manifest.py" "${CMAKE_CURRENT_BINARY_DIR}/frameeyeosc-panel.vrmanifest" "${CMAKE_INSTALL_FULL_BINDIR}/frameeyeosc-panel" "${PANEL_BACKEND}" "${PANEL_CONTROL_SOCKET}" "${PANEL_CONFIG}" COMMAND_ERROR_IS_FATAL ANY)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/frameeyeosc-panel.vrmanifest" DESTINATION ${CMAKE_INSTALL_DATADIR}/frameeyeosc-panel)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(panel-tests tests/core_test.cpp)
  target_link_libraries(panel-tests PRIVATE panel_core)
  target_compile_definitions(panel-tests PRIVATE FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/ipc-v1-subset.jsonl")
  add_test(NAME core COMMAND panel-tests)
  add_test(NAME manifest COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/manifest_test.py")
  add_test(NAME configure-contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/configure_test.py" "${CMAKE_COMMAND}" "${CMAKE_CURRENT_SOURCE_DIR}" "${json_SOURCE_DIR}" "${openvr_SOURCE_DIR}")
  add_executable(overlay-tests tests/overlay_test.cpp)
  target_link_libraries(overlay-tests PRIVATE panel_core)
  target_compile_definitions(overlay-tests PRIVATE FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/ipc-v1-subset.jsonl")
  add_test(NAME overlay-runtime COMMAND overlay-tests)
  add_library(panel_core_test src/child.cpp src/ipc_client.cpp src/overlay.cpp)
  target_link_libraries(panel_core_test PUBLIC nlohmann_json::nlohmann_json openvr_api)
  target_compile_options(panel_core_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_definitions(panel_core_test PUBLIC PANEL_CHILD_TEST_HOOK)
  add_executable(panel-driver tests/driver.cpp)
  target_link_libraries(panel-driver PRIVATE panel_core_test)
  add_test(NAME unix-process COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/process_test.py" "$<TARGET_FILE:panel-driver>")
  set_tests_properties(unix-process PROPERTIES TIMEOUT 60)
endif()
